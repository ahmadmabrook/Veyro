def gate_summary(conn, label):
    rep = gates.run_rls_gate(conn, ti.RUNTIME_ROLE, required_tables=[str(ti.ACCESS_GRANT)],
                             manifest={"version": 2, "platform_scope_relations": {},
                                       "reviewed_security_definer_functions": {}},
                             resolver_root=REPO)
    out("GATE", {"case": label, "status": rep.status,
                 "violations": [(v.code, v.subject, v.detail) for v in rep.violations]})
    return rep


def leak_count(rt_dsn, relation, own):
    with psycopg.connect(rt_dsn, autocommit=False) as rt:
        rt.execute("SELECT set_config('app.tenant_id', %s, true)", (str(own),))
        n = rt.execute(f"SELECT count(*) FROM {relation} WHERE tenant_id <> %s", (own,)).fetchone()[0]
        rt.rollback()
        return n


def main():
    with disposable_postgres() as pg:
        with psycopg.connect(pg.superuser_dsn, autocommit=True) as admin:
            # Facts on a stock, untouched database first.
            for k, q in FACTS.items():
                out("FACT", {"fact": k, "rows": rows(admin, q)})

            rt_password = ti.provision_roles(admin)
            ti.create_tenant_table(admin, ti.ACCESS_GRANT, ti.ACCESS_GRANT_COLUMNS)
            a, b = uuid4(), uuid4()
            ti.seed(admin, ti.ACCESS_GRANT, a, [ti.access_grant_row(1)])
            ti.seed(admin, ti.ACCESS_GRANT, b, [ti.access_grant_row(2), ti.access_grant_row(3),
                                                ti.access_grant_row(4)])
            rt_dsn = pg.dsn(ti.RUNTIME_ROLE, rt_password)
            gate_summary(admin, "P0_clean_stock_with_pin")

            # Literal D1.5 (pin disabled) on the same clean database.
            saved = dict(gates.PINNED_INFORMATION_SCHEMA_PUBLIC)
            gates.PINNED_INFORMATION_SCHEMA_PUBLIC.clear()
            rep = gate_summary(admin, "P0_clean_stock_literal_D1.5_no_pin")
            out("FACT", {"fact": "literal_D1.5_violation_count", "rows": len(rep.violations)})
            gates.PINNED_INFORMATION_SCHEMA_PUBLIC.update(saved)

            def case(label, setup, teardown, leak_rel=None):
                for s in setup:
                    admin.execute(s)
                try:
                    rep = gate_summary(admin, label)
                    if leak_rel:
                        out("LEAK", {"case": label, "cross_tenant_rows_seen_by_runtime":
                                     leak_count(rt_dsn, leak_rel, a)})
                    return rep
                finally:
                    for s in teardown:
                        admin.execute(s)

            # H1: non-stock view created in information_schema, PUBLIC SELECT (superuser-owned => bypasses RLS)
            case("H1_new_view_in_information_schema_PUBLIC_SELECT",
                 ["CREATE VIEW information_schema.zz_leak AS SELECT * FROM harness.access_grant",
                  "GRANT SELECT ON information_schema.zz_leak TO PUBLIC"],
                 ["DROP VIEW information_schema.zz_leak"], "information_schema.zz_leak")
            # H2: same view granted directly to runtime
            case("H2_new_view_in_information_schema_runtime_SELECT",
                 ["CREATE VIEW information_schema.zz_leak AS SELECT * FROM harness.access_grant",
                  f"GRANT SELECT ON information_schema.zz_leak TO {ti.RUNTIME_ROLE}"],
                 ["DROP VIEW information_schema.zz_leak"])
            # H3: new table in information_schema with PUBLIC SELECT
            case("H3_new_table_in_information_schema_PUBLIC_SELECT",
                 ["CREATE TABLE information_schema.zz_t AS SELECT * FROM harness.access_grant",
                  "GRANT SELECT ON information_schema.zz_t TO PUBLIC"],
                 ["DROP TABLE information_schema.zz_t"], "information_schema.zz_t")
            # H4: stock relation, PUBLIC UPDATE added
            case("H4_stock_sql_features_PUBLIC_UPDATE",
                 ["GRANT UPDATE ON information_schema.sql_features TO PUBLIC"],
                 ["REVOKE UPDATE ON information_schema.sql_features FROM PUBLIC"])
            # H5: CREATE on the schema to PUBLIC
            case("H5_information_schema_PUBLIC_CREATE",
                 ["GRANT CREATE ON SCHEMA information_schema TO PUBLIC"],
                 ["REVOKE CREATE ON SCHEMA information_schema FROM PUBLIC"])
            # H6: column-level PUBLIC SELECT on a stock table
            case("H6_stock_column_PUBLIC_UPDATE",
                 ["GRANT UPDATE (comments) ON information_schema.sql_features TO PUBLIC"],
                 ["REVOKE UPDATE (comments) ON information_schema.sql_features FROM PUBLIC"])
            # H7: new SECURITY DEFINER function in information_schema (PUBLIC EXECUTE by default)
            case("H7_new_secdef_function_in_information_schema",
                 ["CREATE FUNCTION information_schema.zz_count() RETURNS bigint SECURITY DEFINER "
                  "LANGUAGE sql AS 'SELECT count(*) FROM harness.access_grant'"],
                 ["DROP FUNCTION information_schema.zz_count()"])
            # H8: same in pg_catalog (independent of the pin)
            case("H8_new_secdef_function_in_pg_catalog",
                 ["CREATE FUNCTION pg_catalog.zz_count() RETURNS bigint SECURITY DEFINER "
                  "LANGUAGE sql AS 'SELECT count(*) FROM harness.access_grant'"],
                 ["DROP FUNCTION pg_catalog.zz_count()"])
            with psycopg.connect(pg.superuser_dsn, autocommit=True) as c2:
                c2.execute("CREATE FUNCTION pg_catalog.zz_count() RETURNS bigint SECURITY DEFINER "
                           "LANGUAGE sql AS 'SELECT count(*) FROM harness.access_grant'")
                with psycopg.connect(rt_dsn, autocommit=False) as rt:
                    rt.execute("SELECT set_config('app.tenant_id', %s, true)", (str(a),))
                    out("LEAK", {"case": "H8_runtime_calls_pg_catalog.zz_count",
                                 "rows_counted_all_tenants": rt.execute("SELECT pg_catalog.zz_count()").fetchone()[0],
                                 "rows_visible_directly": rt.execute("SELECT count(*) FROM harness.access_grant").fetchone()[0]})
                c2.execute("DROP FUNCTION pg_catalog.zz_count()")
            # H9: revoke stock PUBLIC SELECT (tightening) — should stay PASS
            case("H9_tightening_revoke_stock_PUBLIC_SELECT",
                 ["REVOKE SELECT ON information_schema.tables FROM PUBLIC"],
                 ["GRANT SELECT ON information_schema.tables TO PUBLIC"])
            gate_summary(admin, "P0_clean_after_all_cases")


if __name__ == "__main__":
    main()
